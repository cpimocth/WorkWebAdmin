<script>
    let rawData = [];
    const areaSelector = document.getElementById('areaSelector');
    const statusFilter = document.getElementById('statusFilter');
    const tableBody = document.querySelector('#dataTable tbody');
    const searchInput = document.getElementById('searchInput');

    async function fetchData() {
        try {
            // ดึงไฟล์ JSON ที่สร้างจาก Python (เปลี่ยนชื่อตามต้องการ)
            const response = await fetch('my_data.json'); 
            const res = await response.json();
            rawData = res.data; 
            if (res.metadata) document.getElementById('dateDisplay').textContent = res.metadata.display_date;
            init();
        } catch (e) { console.error(e); }
    }

    function determineColor(value, threshold) {
        const v = Number(value), t = Number(threshold);
        if (v === 0) return 'is-red';         
        if (v > t)   return 'is-blue';         
        if (v === t) return 'is-green';       
        return 'is-yellow';                   
    }

    function init() {
        if (!rawData.length) return;
        const skip = ["ปี/เดือน","CODE7", "รายการ", "กำหนดให้เก็บ", "รายละเอียด"];
        const cols = Object.keys(rawData[0]).filter(k => !skip.includes(k));
        areaSelector.innerHTML = '<option value="" disabled selected>เลือกพื้นที่...</option>';
        cols.forEach(c => {
            let opt = document.createElement("option");
            opt.value = c; opt.textContent = c;
            areaSelector.appendChild(opt);
        });
        renderTable();
    }

    function renderTable() {
        const area = areaSelector.value, status = statusFilter.value, search = searchInput.value.toLowerCase();
        if (!area) return;

        let html = '', stats = { total: 0, blue: 0, green: 0, yellow: 0, red: 0 };

        rawData.forEach(item => {
            const code = String(item.CODE7 || "").toLowerCase();
            const name = String(item.รายการ || "").toLowerCase();
            const val = item[area] || 0, goal = item.กำหนดให้เก็บ || 0;
            const cls = determineColor(val, goal);

            if (name.includes(search) || code.includes(search)) {
                stats.total++;
                stats[cls.replace('is-', '')]++;

                if (status === 'all' || cls === status) {
                    let stTxt = (cls==='is-blue')?'เกิน':(cls==='is-green')?'ครบ':(cls==='is-red')?'ขาด':'ไม่ครบ';
                    html += `<tr>
                        <td><code>${item.CODE7}</code></td>
                        <td><b>${item.รายการ}</b></td>
                        <td style="text-align:center;">${goal}</td>
                        <td style="text-align:center; font-weight:700;">${val}</td>
                        <td style="text-align:center;"><span class="status-badge ${cls}">${stTxt}</span></td>
                        <td style="font-size:0.8rem; color:#7f8c8d;">${item.รายละเอียด || "-"}</td>
                    </tr>`;
                }
            }
        });

        document.getElementById('sumTotal').textContent = stats.total;
        document.getElementById('sumBlue').textContent = stats.blue;
        document.getElementById('sumGreen').textContent = stats.green;
        document.getElementById('sumYellow').textContent = stats.yellow;
        document.getElementById('sumRed').textContent = stats.red;

        const prg = stats.total > 0 ? Math.round(((stats.green + stats.blue) / stats.total) * 100) : 0;
        const pBar = document.getElementById('mainProgressBar');
        pBar.style.width = prg + '%';
        pBar.textContent = `ความคืบหน้า: ${prg}%`;
        pBar.style.background = prg < 50 ? 'var(--danger)' : prg < 80 ? 'var(--warning)' : 'var(--success)';

        tableBody.innerHTML = html || '<tr><td colspan="6" style="text-align:center; padding:40px;">ไม่พบข้อมูล</td></tr>';
    }

    function exportToCSV() {
        const area = areaSelector.value;
        if (!area) return alert("กรุณาเลือกพื้นที่ก่อน");

        let csv = ['รหัส,รายการ,เป้าหมาย,เก็บจริง,สถานะ,รายละเอียด'];
        document.querySelectorAll("#dataTable tbody tr").forEach(row => {
            const cols = row.querySelectorAll("td");
            if (cols.length > 1) {
                const data = Array.from(cols).map(c => `"${c.innerText.replace(/"/g, '""')}"`);
                csv.push(data.join(","));
            }
        });

        const blob = new Blob(["\ufeff" + csv.join("\n")], { type: "text/csv;charset=utf-8;" });
        const link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.download = `CPI_Report_${area}.csv`;
        link.click();
    }

    fetchData();
</script>
